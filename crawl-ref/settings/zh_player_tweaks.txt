function withColor(color, str)

-- 统一输出。用 ASCII 标记替代原版 emoji（见 CONFIG 注释）。

##
## AnnounceDamage.lua —— 每回合一行血蓝变化
################################################################################
##
## 与原版配置的差别（都是为了让消息流更干净，而不是更啰嗦）：
##
##  1. 只输出一行，形如  生命[-3] 法力[-1]
##     原版一次掉血最多会连出三条：浓缩行 + "你受到 3 点伤害，现在 20/23" +
##     "你的最大生命值增加了…"。血条和游戏日志已经说过的事不再重复。
##  2. 不再用 "\n" 前缀。原版每次掉血都会先空一行，在小消息窗里很浪费。
##  3. 删掉"满血恢复"播报（HPFull/MPFull）。回到满血时数字本身就是满的，
##     而这条消息在再生、治疗、休息时反复出现，是最主要的噪音源。
##  4. 删掉回血/掉蓝长句，只保留带符号的数字。
##  5. 最大生命变化合并进同一行（"最大生命[-5]"），不再单独成句，也不再被误判为受伤。
##  6. 删掉 emoji 前缀；改为把"生命"标签按当前血量比例上色，
##     既保留"余光看危险"的价值，又不占额外字符宽度。
##     标签文字取当前界面译文（crawl.t_）；本模板仅支持中文界面。
##  7. 重伤标记与同一行合并（"重伤!!"），并交由 force_more 触发急停，
##     不再额外打印第二条消息。
################################################################################
{
ZH_TWEAKS = ZH_TWEAKS or {}
local prev_hp, prev_hpmax = 0, 0
local prev_mp, prev_mpmax = 0, 0

-- 标签取当前中文界面译文：生命 / 法力 / 最大生命。
-- 这样译文调整时 rc 不需要跟着改。
local LABEL_HP    = crawl.t_("HP")
local LABEL_MP    = crawl.t_("MP")
local LABEL_MAXHP = crawl.t_("Max HP")

-- 重伤标记：故意用半角 "!!"，中文消息用的是全角"！"，不会误配。
local MASSIVE_MARK = "重伤!!"

local function hp_colour(hp, hpmax)
  if hpmax <= 0 then return COLORS.lightgrey end
  local pct = hp * 100 / hpmax
  if pct <= 25 then return COLORS.red
  elseif pct <= 50 then return COLORS.lightred
  elseif pct <= 75 then return COLORS.yellow
  end
  return COLORS.lightgrey
end

local function delta_text(n)
  return withColor(n < 0 and COLORS.red or COLORS.green,
                   string.format("[%+d]", n))
end

function announce_damage()
  local hp, hpmax = you.hp()
  local mp, mpmax = you.mp()

  -- 新角色第一回合、死亡回合都不播报
  if prev_hp > 0 and hp > 0 then
    local d_hp    = hp - prev_hp
    local d_hpmax = hpmax - prev_hpmax
    local d_mp    = mp - prev_mp

    local parts = {}

    local show_hp = d_hp < 0 or d_hp >= CONFIG.min_gain_to_show
    if d_hp ~= 0 and show_hp then
      parts[#parts + 1] = withColor(hp_colour(hp, hpmax), LABEL_HP)
                          .. delta_text(d_hp)
    end
    if d_hpmax ~= 0 then
      parts[#parts + 1] = withColor(COLORS.lightblue,
                                    string.format("%s[%+d]", LABEL_MAXHP, d_hpmax))
    end
    local show_mp = d_mp < 0 or d_mp >= CONFIG.min_gain_to_show
    if d_mp ~= 0 and show_mp then
      parts[#parts + 1] = withColor(COLORS.cyan, LABEL_MP)
                          .. delta_text(d_mp)
    end

    -- 净伤害只扣除"上限下降"那部分（如死灵系上限被抽走），
    -- 上限上升（升级/增益）绝不能算成受伤。
    local net_damage = -d_hp + math.min(0, d_hpmax)
    if net_damage > 0 and hpmax > 0
       and net_damage * 100 / hpmax >= CONFIG.massive_damage_pct then
      parts[#parts + 1] = withColor(COLORS.lightred, MASSIVE_MARK)
    end

    if #parts > 0 then
      crawl.mpr(table.concat(parts, " "))
    end
  end

  prev_hp, prev_hpmax = hp, hpmax
  prev_mp, prev_mpmax = mp, mpmax
end
}

##
## PickupEquipment.lua
################################################################################
##
## 装备拾取沿用原版策略，按当前 trunk 的槽名和 subtype 接口适配：
##   自动拾取回调收到的名字来自 _autopickup_item_name()（内部强制英文），
##   item.class()/subtype()/ego() 与槽名使用英文；item.name() 使用界面语言。
## 仅删除原文件里的调试注释块。
################################################################################
{
local favor_egos = false
local favor_plus = false

local function is_branded_ego(item)
  return item.branded or item.ego()
end

local function is_magical(item)
  return is_branded_ego(item) or item.artefact
end

local function should_pickup(cur, i2)
  -- 神赐装备永远捡
  if i2.god_gift then return true end

  -- 空槽位：捡
  if cur == nil then return true end

  local higher_plus = i2.plus ~= nil and i2.plus > (cur.plus or 0)
  local more_magical = not is_magical(cur) and is_magical(i2)

  -- 同名物品：附魔更高或更"魔法"就捡
  if cur.name("qual") == i2.name("qual") then
    if higher_plus or more_magical then return true end
  end

  local is_dragon_scales = string.find(cur.subtype(false), "dragon scale")
  -- 已穿魔法装备或龙鳞：不再捡（注意：跨龙鳞种类的升级会因此被跳过）
  if is_magical(cur) or is_dragon_scales then return end

  if favor_egos and is_magical(i2) then return true end
  if favor_plus and higher_plus then return true end

  return false
end

local item_subtype_equip_slot = { }
item_subtype_equip_slot.cloak  = "Cloak"
item_subtype_equip_slot.helmet = "Helmet"
item_subtype_equip_slot.gloves = "Gloves"
item_subtype_equip_slot.boots  = "Boots"
item_subtype_equip_slot.offhand = "Offhand"
item_subtype_equip_slot.barding = "Barding"

local function equip_slot(item_sub_type)
  if item_sub_type == "body" then
    -- webtiles 认 "Body Armour"，本地认 "Armour"
    return items.equipped_at("Body Armour") or items.equipped_at("Armour")
  end
  local slot = item_subtype_equip_slot[item_sub_type]
  if slot then return items.equipped_at(slot) end
end

-- 盾牌拾取策略：true=拾取 / false=不拾取 / nil=不表态（交回游戏规则）。
-- 单独抽成函数，方便玩家改，也方便自测直接断言。
-- 注意：Lua 返回 true/false 会覆盖 autopickup_exceptions 与 autopickup 设置，
-- `\` 已识别物品菜单不列出盾牌；请使用 CONFIG 开关。
function zh_tweaks_shield_decision(current_weapon, shield, shield_skill)
  if not CONFIG.autopickup_shields then return false end
  if current_weapon and current_weapon.hands == 2 then
    return false
  end
  if (shield_skill or 0) <= 3 then return false end
  local current = equip_slot("offhand")
  -- 玩家已选宝珠时，不自动收集盾牌来替代它。
  if current and current.subtype(false) == "orb" then return false end
  if should_pickup(current, shield) then return true end
  return nil
end

local function pickup_equipment(it, name)
  local class = it.class(true)

  -- 禁止物品（由 AutoInscribe 的 forbidden 铭刻标记）
  if string.match(name, "forbidden") then return end
  if it.is_useless then return end

  -- 神器永远捡
  if it.artefact then return true end

  if class == "weapon" then
    local currentWeapon = items.equipped_at("weapon")

    -- 徒手角色不捡武器
    if currentWeapon == nil then
      if it.god_gift then return true end
      return false
    end

    if should_pickup(currentWeapon, it) then return true end

  elseif class == "armour" then
    local sub_type = it.subtype()

    if sub_type == "gloves" and you.has_claws() > 0 then return end

    if sub_type == "offhand" then
      -- 宝珠也是 offhand，但不是盾；交回游戏默认拾取规则。
      if it.subtype(false) == "orb" then return end
      return zh_tweaks_shield_decision(items.equipped_at("Weapon"), it,
                                   you.skill("Shields"))
    end

    -- 未知槽位不应被误当作空槽位而强制拾取。
    if sub_type ~= "body" and not item_subtype_equip_slot[sub_type] then return end

    if should_pickup(equip_slot(sub_type), it) then return true end
  end

  return
end

-- rc 可在选角后或读档时重读；保持稳定入口，仅替换实现。
ZH_TWEAKS.pickup_impl = pickup_equipment
ZH_TWEAKS.pickup_wrapper = ZH_TWEAKS.pickup_wrapper or function(it, name)
  return ZH_TWEAKS.pickup_impl(it, name)
end
local registered = false
for _, func in ipairs(chk_force_autopickup) do
  if func == ZH_TWEAKS.pickup_wrapper then registered = true end
end
if not registered then add_autopickup_func(ZH_TWEAKS.pickup_wrapper) end
}

## NoteVersion / TurnReady.lua
################################################################################
{
local didRun = false
function note_version()
  if didRun then return end
  crawl.take_note("zh_player_tweaks.txt [v1.3]")
  didRun = true
end
}

{
-- include 后另定义 ready() 的玩家，请在自己的 ready 中调用 zh_tweaks_ready()。
-- include 前的 ready 自动保留。稳定包装器避免重读时包装器互相套娃。
function zh_tweaks_ready()
  if ZH_TWEAKS.dispatching and ZH_TWEAKS.did_ready then return end
  if ZH_TWEAKS.dispatching then ZH_TWEAKS.did_ready = true end
  announce_damage()
  note_version()
end
if ready ~= ZH_TWEAKS.ready_wrapper then ZH_TWEAKS.user_ready = ready end
ZH_TWEAKS.ready_wrapper = ZH_TWEAKS.ready_wrapper or function()
  if ZH_TWEAKS.dispatching then return end
  ZH_TWEAKS.dispatching, ZH_TWEAKS.did_ready = true, false
  local ok, err = pcall(function()
    if ZH_TWEAKS.user_ready then ZH_TWEAKS.user_ready() end
    zh_tweaks_ready()
  end)
  ZH_TWEAKS.dispatching, ZH_TWEAKS.did_ready = false, false
  if not ok then error(err) end
end
ready = ZH_TWEAKS.ready_wrapper
}

#
# 一、为什么删掉了大部分原配置（约 1500 行）
#
#   原版配置的体积主要来自三块英文规则，在汉化版里全部失效且无法简单复用：

#   1. Messages.rc 的约 1000 条 message_colour，以及 ItemColors.rc 的

#      已经提供了随版本维护的中文规则；原文件开头的 "msc =" / "menu ="
#      会把这些中文规则一起清空，所以这里干脆不写这两节。
#
#   2. Morgue.rc 的 100 多条 note_messages（含逐个非随机神器的名字清单）。

#      已按上面的设计重写为中文，并按 more/stop 两级拆分。
#
# 二、保留的部分为什么可以原样用
#
#   自动拾取（autopickup_exceptions）、自动铭刻（autoinscribe）、
#   能力/法术字母位，匹配的都是代码里强制英文的内部名；Lua 侧
#   item.class()/subtype()/ego()、you.race()/god()/skill()、
#   items.equipped_at() 按英文槽名查装备，item.name() 返回界面语言。
#   自动铭刻/自动拾取例外匹配英文；装备槽位接口需随 trunk 维护。
#
#   唯一必须翻译的是 gear_slot：它匹配物品显示名（中文）。
#
# 三、维护：添加中文消息规则的方法
#
#   1. 在 dat/i18n/zh/source.txt 里查这条消息的实际译文（英文是键，中文是值）；
#   2. 只取关键词或固定短语，不要整句（译文会随评审改进而变化）；
#   3. 先判断该用 more 还是 stop：
#        · 不按键就会漏掉、且代价不可逆 → more
#        · 只要中止跑动即可 → stop
#   4. 用 crawl.message_filter 自测一下再提交：
#        local f = crawl.message_filter("你的规则", -1)
#        f:matches("你预期的译文", -1)      -- 应为 true
#        f:matches("不该命中的无关消息", -1) -- 应为 false
#
# 四、在 include 之后追加规则时注意别名作用域
#
#   本文件里的 more / stop / ignore / ai / ae 都是 := 定义的别名，
#   而 crawl 只在本文件内保留它们：include 结束后别名会被撤销
#   （initfile.cc 的 include() 会 unwind 被包含文件里定义的别名）。
#   所以：
#
#     include = zh_player_tweaks.txt
#     more += 你进入了佐特陷阱        ← 无效（more 已不存在）
#     force_more_message += 你进入了佐特陷阱   ← 正确写法
#
#   要在本文件之外追加规则，请使用全名 force_more_message /
#   runrest_stop_message / runrest_ignore_message / autoinscribe /
#   autopickup_exceptions。
#

#   新增石化前兆、神祇保命、赎罪警告三个 more；不扩大高频 stop。
#   译文键见各规则的 EN 注释；保命规则保留全角感叹号，以排除冰甲说明。

#      "ae += useless_item"（以及 forbidden / <evoker / <good_item）一起删了。
#      法术书在 autopickup 的 "+" 字符里，失去 useless_item 例外后就会被捡。
#      现在不再清空，只在默认规则之后追加本文件的补充项。

#   2. 盾牌始终被自动拾取、游戏内关不掉：
#      原逻辑在"盾技 > 3 且没持盾"时返回 true。Lua 返回布尔值会**覆盖**

#        · Lua 返回 true  = 强制拾取，压过 ae 例外与 autopickup 字符
#        · Lua 返回 false = 强制不捡，同样压过 ae
